       terraform {
         required_version = ">= 1.5"
         required_providers {

             version = "~> 7.19"

         region  = var.region

       module "vpc" { source = "./modules/vpc" ... }
       module "cloud_sql" { source = "./modules/cloud-sql", depends_on = [module.vpc] ... }
       module "memorystore" { source = "./modules/memorystore", depends_on = [module.vpc] ... }
       module "pubsub" { source = "./modules/pubsub" ... }

    - Output: connection_name, private_ip, database_name, user_name

      - `dead_letter_policy` pointing to dead-letter topic

    - Output: topic_names map, subscription_names map

    **4. Create `scripts/connectivity-check.ts`:**

    The script tests reachability of each infrastructure component and reports pass/fail with clear diagnostic output.

    ```typescript
    #!/usr/bin/env tsx
    import pg from 'pg';
    import Redis from 'ioredis';
    import { PubSub } from '@google-cloud/pubsub';

    interface CheckResult {
      name: string;
      status: 'PASS' | 'FAIL' | 'SKIP';
      message: string;
      durationMs: number;
    }

    const results: CheckResult[] = [];

    async function checkPostgres(): Promise<void> {
      const start = Date.now();
      const url = process.env.DATABASE_URL;
      if (!url) {
        results.push({ name: 'PostgreSQL', status: 'SKIP', message: 'DATABASE_URL not set', durationMs: 0 });
        return;
      }
      const client = new pg.Client({ connectionString: url });
      await client.connect();
      const res = await client.query('SELECT current_database(), version()');
      await client.end();
      results.push({
        name: 'PostgreSQL',
        status: 'PASS',
        message: `Connected to ${res.rows[0].current_database}`,
        durationMs: Date.now() - start,
      });
    }

    async function checkRedis(): Promise<void> {
      const start = Date.now();
      const url = process.env.REDIS_URL;
      if (!url) {
        results.push({ name: 'Redis', status: 'SKIP', message: 'REDIS_URL not set', durationMs: 0 });
        return;
      }
      const redis = new Redis(url);
      const pong = await redis.ping();
      redis.disconnect();
      results.push({
        name: 'Redis',
        status: pong === 'PONG' ? 'PASS' : 'FAIL',
        message: pong === 'PONG' ? 'PONG received' : `Unexpected response: ${pong}`,
        durationMs: Date.now() - start,
      });
    }

    async function checkPubSub(): Promise<void> {
      const start = Date.now();
      const emulatorHost = process.env.PUBSUB_EMULATOR_HOST;
      const projectId = process.env.GCP_PROJECT_ID || 'claw-army-dev';

      if (!emulatorHost) {
        // Try real GCP Pub/Sub
        try {
          const pubsub = new PubSub({ projectId });
          const [topics] = await pubsub.getTopics();
          results.push({
            name: 'Pub/Sub (GCP)',
            status: 'PASS',
            message: `${topics.length} topics found`,
            durationMs: Date.now() - start,
          });
        } catch (e: any) {
          results.push({
            name: 'Pub/Sub (GCP)',
            status: 'FAIL',
            message: e.message,
            durationMs: Date.now() - start,
          });
        }
        return;
      }

      // Using emulator
      const pubsub = new PubSub({ projectId });
      const [topics] = await pubsub.getTopics();
      results.push({
        name: 'Pub/Sub (emulator)',
        status: 'PASS',
        message: `Emulator reachable at ${emulatorHost}, ${topics.length} topics`,
        durationMs: Date.now() - start,
      });
    }
    ```

    The script should:
    - Run each check independently (one failure does not prevent others)
    - Print a formatted summary table at the end
    - Exit with code 0 if all non-skipped checks pass, exit 1 if any fail
    - Use colored output (green PASS, red FAIL, yellow SKIP) if terminal supports it
    - Accept `--local` flag to auto-set env vars for docker-compose.dev.yml defaults
