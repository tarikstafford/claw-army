```typescript

const LLM_INPUT_RATE_CENTS_PER_M = Number(process.env.LLM_INPUT_RATE_CENTS_PER_M ?? 15);
const LLM_OUTPUT_RATE_CENTS_PER_M = Number(process.env.LLM_OUTPUT_RATE_CENTS_PER_M ?? 60);
// Bot hourly rate in cents (for bot-hours cost component)
const BOT_HOURLY_RATE_CENTS = Number(process.env.BOT_HOURLY_RATE_CENTS ?? 100);
```

**Redis client:** Module-level IORedis instance from `REDIS_URL`.

**Pub/Sub client:**
```typescript
const pubsub = new PubSub({
  projectId: process.env.GCP_PROJECT_ID ?? 'claw-local',
```

**Atomic budget enforcement Lua script:**
```typescript
const BUDGET_ENFORCE_SCRIPT = `
local spend_key = KEYS[1]
local cap_key = KEYS[2]
local amount = tonumber(ARGV[1])
local cap = tonumber(redis.call('GET', cap_key))
if cap == nil then
  local new_total = redis.call('INCRBY', spend_key, amount)
  return {new_total, 0}
end
local new_total = redis.call('INCRBY', spend_key, amount)
if new_total > cap then
  return {new_total, 1}
end
return {new_total, 0}
`;
```

**Core function: `enforceAtomicBudget(executionId, amountCents)`**
Returns `{ newTotalCents: number; capExceeded: boolean }`.
Calls `redis.eval(BUDGET_ENFORCE_SCRIPT, 2, 'budget:spend:${executionId}', 'budget:cap:${executionId}', amountCents.toString())`.
Cast result as `[number, number]` — `[0]` = new total, `[1]` = 0 (ok) or 1 (exceeded).

**Message deduplication:**
Before processing any message, check Redis: `redis.set('processed:${message.id}', '1', 'EX', 86400, 'NX')`. If returns null (key already existed), ACK and skip. This prevents double-counting from Pub/Sub at-least-once redelivery.

**Billing event persistence: `writeBillingEvent(params)`**
Insert a row into `billingEvents` table via Drizzle:
```typescript
async function writeBillingEvent(params: {
  executionId: string;
  botId?: string;
  eventType: 'bot_started' | 'bot_stopped' | 'tool_invoked' | 'execution_completed' | 'budget_exceeded';
  amountCents?: number;
  tokenCount?: number;
  metadata?: Record<string, unknown>;
}): Promise<void> {
  await db.insert(billingEvents).values({
    executionId: params.executionId,
    botId: params.botId,
    eventType: params.eventType,
    amountCents: params.amountCents,
    tokenCount: params.tokenCount,
    metadata: params.metadata,
  });
}
```

**Bot-hours calculation: `recordBotHours(botId, executionId)`**
Called when `bot_stopped` event arrives:
1. Query `bots` table for `startedAt` and `stoppedAt`.
2. If both exist, calculate `wallClockMs = stoppedAt.getTime() - startedAt.getTime()`.
3. Calculate `botHours = wallClockMs / (1000 * 60 * 60)`.
4. Insert into `telemetry` table: `{ executionId, botId, metricName: 'bot_hours', metricValue: botHours.toFixed(6) }`.

**Token cost calculation: `calculateTokenCost(promptTokens, completionTokens)`**
Returns cost in integer cents:
```typescript
function calculateTokenCost(promptTokens: number, completionTokens: number): number {
  return Math.round(
    (promptTokens * LLM_INPUT_RATE_CENTS_PER_M + completionTokens * LLM_OUTPUT_RATE_CENTS_PER_M) / 1_000_000
  );
}
```

**Event handler for billing-events topic: `handleBillingMessage(message)`**
Parse the message data as JSON. These messages have `type: 'billing_event'`. Based on the `eventType` field:

1. **`eventType: 'tool_invoked'`:**
   - If the event has `tokenCount` and `amountCents` (LLM call), call `enforceAtomicBudget(executionId, amountCents)`.
   - `writeBillingEvent({ executionId, botId, eventType: 'tool_invoked', amountCents, tokenCount })`
   - If `capExceeded`: call `handleBudgetExceeded(executionId)`.

2. **`eventType: 'execution_completed'`:**
   - `writeBillingEvent({ executionId, eventType: 'execution_completed' })`

**SEPARATE event handler for bot-lifecycle topic: `handleBotLifecycleMessage(message)`**
Parse the message data as JSON. These messages have `type: 'bot_started'` or `type: 'bot_stopped'` (NOT `type: 'billing_event'`). Based on the `type` field:

1. **`type: 'bot_started'`:**
   - `writeBillingEvent({ executionId, botId, eventType: 'bot_started' })`

2. **`type: 'bot_stopped'`:**
   - `writeBillingEvent({ executionId, botId, eventType: 'bot_stopped' })`
   - `recordBotHours(botId, executionId)`

**CRITICAL:** The bot-lifecycle topic publishes events with `type: 'bot_started'`/`type: 'bot_stopped'` (as published by bot-orchestrator's `publishBotStarted`/`publishBotStopped`). These are NOT wrapped in a `billing_event` envelope. The billing-events subscription handler (`handleBillingMessage`) and the bot-lifecycle subscription handler (`handleBotLifecycleMessage`) MUST be distinct functions with different dispatch logic. Routing both subscriptions through `handleBillingMessage` will silently drop bot lifecycle events because the `type` field won't match `'billing_event'`.

**Budget exceeded handler: `handleBudgetExceeded(executionId)`**
1. Get the cap from Redis: `redis.get('budget:cap:${executionId}')`.
2. Get the current spend: `redis.get('budget:spend:${executionId}')`.
3. `writeBillingEvent({ executionId, eventType: 'budget_exceeded', amountCents: Number(currentSpend) })`.
4. Publish `budgetExceeded` event via `publishBudgetExceeded()`.
5. Stop all bots in the execution: `getBotsForExecution(executionId)` then `stopBot()` each.
6. Transition execution to 'stopped': `transitionExecution(executionId, 'running', 'stopped')`.

**Subscriber lifecycle:**
```typescript
export function startBillingEngine(): { shutdown: () => Promise<void> } {
  const subName = process.env.BILLING_SUBSCRIPTION ?? 'billing-events-sub';
  const subscription = pubsub.subscription(subName);

  subscription.on('message', async (message: Message) => {
    // Deduplication
    const isNew = await redis.set(`processed:${message.id}`, '1', 'EX', 86400, 'NX');
    if (!isNew) {
      message.ack();
      return;
    }
    try {
      await handleBillingMessage(message);
      message.ack();
    } catch (err) {
      console.error('[billing-engine] Failed to process message:', err);
      message.nack();
    }
  });

  subscription.on('error', (err) => {
    console.error('[billing-engine] Subscription error (non-fatal):', err);
  });

  console.log('[billing-engine] Started, listening on subscription:', subName);

  return {
    shutdown: async () => {
      await subscription.close();
    },
  };
}
```
