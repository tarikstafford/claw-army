- `botId` (string, uuid format)

Query parameters (optional):
- `limit` (integer, min 1, max 100, default 50)

Query logic:
- Select from `decisionTraces` WHERE `botId = :botId`
- Select fields: id, decisionType, directiveReferenced, attributionConfidence, outcome, decidedAt, executionId

}

// Phase 39 — Decision Traces (SOUL-02)

export async function getBotDecisionTraces(
  botId: string,
  params: { limit?: number; offset?: number } = {}
): Promise<DecisionTracesResponse> {
  const query = new URLSearchParams();
  if (params.limit != null) query.set('limit', String(params.limit));
  if (params.offset != null) query.set('offset', String(params.offset));
  const qs = query.toString();
  return apiFetch(`${BASE}/decision-traces/${botId}${qs ? `?${qs}` : ''}`);
}

// Phase 39 — Negative Signals (SOUL-03)

export async function getNegativeSignals(

  if (params.limit != null) query.set('limit', String(params.limit));
  if (params.offset != null) query.set('offset', String(params.offset));
  const qs = query.toString();
  return apiFetch(`${BASE}/negative-signals${qs ? `?${qs}` : ''}`);
}

Add a "Decision Traces" toggle button near the existing "Soul Inspector" toggle (the `showInspector` button). When clicked, it sets `showTraces = !showTraces` and on first open, fetches traces:

```typescript
function loadTraces(reset = false) {
  if (reset) { tracesOffset = 0; decisionTraces = []; }
  tracesLoading = true;
  tracesError = null;
  getBotDecisionTraces(botId, { limit: 30, offset: tracesOffset })
    .then(res => {
      decisionTraces = reset ? res.traces : [...decisionTraces, ...res.traces];
      tracesTotal = res.total;
      tracesHasMore = res.hasMore;
      tracesLoading = false;
    })
    .catch(err => { tracesError = (err as Error).message; tracesLoading = false; });
}
```
