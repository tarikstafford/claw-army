**SSE client (`services/ui/src/lib/sse.ts`):**

EventSource connection factory:

```typescript
import { browser } from '$app/environment';
import type { ActivityEvent } from './types';

const BASE = import.meta.env.VITE_API_URL ?? 'http://localhost:3001';

export function connectSSE(
  executionId: string,
  onEvent: (event: ActivityEvent) => void,
  onError?: (err: Event) => void,
): (() => void) | null {
  if (!browser) return null;

  const es = new EventSource(`${BASE}/executions/${executionId}/events`);

  // Listen for all known event types
  const eventTypes = [
    'execution_status_changed',
    'task_claimed',
    'task_completed',
    'bot_started',
    'bot_stopped',
    'guardrail_triggered',
    'billing_event',
    'budget_exceeded',
  ];

  for (const type of eventTypes) {
    es.addEventListener(type, (e: MessageEvent) => {
      try {
        const payload = JSON.parse(e.data);
        const isAlert = type === 'guardrail_triggered' || type === 'budget_exceeded';
        onEvent({ ...payload, type, isAlert });
      } catch { /* ignore malformed events */ }
    });
  }

  if (onError) es.onerror = onError;

  // Return cleanup function
  return () => es.close();
}
```

Returns a cleanup function (or null if not browser). Components call this in `$effect` and return the cleanup.
