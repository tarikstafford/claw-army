   ```typescript
   import type { PageServerLoad } from './$types';
   import { redirect } from '@sveltejs/kit';

   export const load: PageServerLoad = async ({ url }) => {
     const connected = url.searchParams.get('connected');
     const error = url.searchParams.get('error');
     const tool = url.searchParams.get('tool');

     // If OAuth callback params present, pass to catalog page via query
     if (connected) {
       throw redirect(303, `/tools/catalog?connected=${encodeURIComponent(connected)}`);
     }
     if (error) {
       const params = new URLSearchParams({ error });
       if (tool) params.set('tool', tool);
       throw redirect(303, `/tools/catalog?${params.toString()}`);
     }

     // Default: redirect to catalog tab
     throw redirect(303, '/tools/catalog');
   };
   ```
