**Create `services/ui/src/lib/components/evolution/LineageTree.svelte` (DASH-03):**

Props: `{ nodes: SoulNode[] }` where SoulNode is `{ id, label, generation, isArchetype, isPioneer }`.
The `nodes` array arrives flat (root first, each subsequent is a child of the previous in the chain). Convert to tree structure for d3-hierarchy.

Script:
```svelte
<script lang="ts">
  import { hierarchy, tree } from 'd3-hierarchy';

  interface SoulNode {
    id: string;
    label: string;
    generation: number;
    isArchetype: boolean;
    isPioneer: boolean;
    children?: SoulNode[];
  }

  let { nodes }: { nodes: SoulNode[] } = $props();
  let selectedNode: SoulNode | null = $state(null);

  const WIDTH = 480;
  const HEIGHT = 240;
  const NODE_R = 10;

  // Convert flat chain to nested tree structure
  function buildTree(chain: SoulNode[]): SoulNode {
    if (chain.length === 0) return { id: 'empty', label: '—', generation: 0, isArchetype: false, isPioneer: false };
    const root = { ...chain[0], children: [] as SoulNode[] };
    let current = root;
    for (let i = 1; i < chain.length; i++) {
      const child = { ...chain[i], children: [] as SoulNode[] };
      current.children = [child];
      current = child;
    }
    return root;
  }

  const treeRoot = $derived(buildTree(nodes));

  const layout = $derived.by(() => {
    const h = hierarchy(treeRoot);
    const t = tree<SoulNode>().size([WIDTH - 40, HEIGHT - 60]);
    t(h);
    return h;
  });

  const descendants = $derived(layout.descendants());
  const links = $derived(layout.links());

  function handleNodeClick(node: SoulNode) {
    selectedNode = selectedNode?.id === node.id ? null : node;
  }
</script>
```

SVG rendering: exact pattern from RESEARCH.md Pattern 3:
- `<svg width={WIDTH} height={HEIGHT} class="lineage-tree" role="img" aria-label="Soul lineage tree">`
- `<g transform="translate(20,30)">`
- Links: `{#each links as link (link.target.data.id)}` -> `<line>` with `--bo-border` stroke, 1px
- Nodes: `{#each descendants as node (node.data.id)}` -> `<g>` with transform, `role="button"`, `tabindex="0"`, onclick/onkeydown handlers
  - `<circle r={NODE_R}>`
  - `<text dy="1.4em" text-anchor="middle" class="node-label">{node.data.label}</text>`
- Node classes: `.archetype` for isArchetype, `.pioneer` for isPioneer

Inline tooltip for selected node:

- Styling: `--bo-card` background, `1px solid var(--bo-bhi)`, `var(--space-md)` padding, `--radius-md`
